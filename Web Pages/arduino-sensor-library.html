LDR1

Photoresistor (LDR)

Reads ambient light on an analog pin and switches an LED on once it gets dark — the "hello world" of analog sensing.

SigA0 ← LDR wiper
OutD13 → LED
Logicvalue < 500 → dark
ldr_sensor.ino
int ldr = A0;
int led = 13;

void setup() {
  pinMode(led, OUTPUT);
  Serial.begin(9600);
}

void loop() {

  int value = analogRead(ldr);

  if (value < 500) {      // Dark
    digitalWrite(led, HIGH);
  } else {                // Bright
    digitalWrite(led, LOW);
  }

  Serial.println(value);

  delay(100);
}

FLX1

Flex Sensor

Maps a raw bend reading to a 0–180° angle, then lights one of four LEDs depending on which quarter of the bend range it falls in.

SigA0 ← divider wiper
Zone 1–4D2 · D3 · D4 · D5
Rangeraw 990–1017 → 0–180°
flex_sensor.ino
int flex = A0;

int led1 = 2;
int led2 = 3;
int led3 = 4;
int led4 = 5;

void setup() {

  Serial.begin(9600);

  pinMode(led1, OUTPUT);
  pinMode(led2, OUTPUT);
  pinMode(led3, OUTPUT);
  pinMode(led4, OUTPUT);

}

void loop() {

  int raw = analogRead(flex);

  int angle = map(raw, 990, 1017, 0, 180);

  angle = constrain(angle, 0, 180);

  Serial.print("Raw Value : ");
  Serial.print(raw);
  Serial.print("    Bend Angle : ");
  Serial.print(angle);
  Serial.println(" Degree");

  digitalWrite(led1, LOW);
  digitalWrite(led2, LOW);
  digitalWrite(led3, LOW);
  digitalWrite(led4, LOW);

  if (angle < 45) {
    digitalWrite(led1, HIGH);
  }
  else if (angle < 90) {
    digitalWrite(led2, HIGH);
  }
  else if (angle < 135) {
    digitalWrite(led3, HIGH);
  }
  else {
    digitalWrite(led4, HIGH);
  }

  delay(200);

}
SOIL1

Soil Moisture Sensor

Buckets the analog reading into four moisture bands — green through red — driving one indicator LED per band.

SigA0 ← soil probe
BandsD2 green · D3 yellow · D4 orange · D5 red
Thresholds≤250 · ≤500 · ≤750 · >750
soil_moisture.ino
int soil = A0;

int green = 2;
int yellow = 3;
int orange = 4;
int red = 5;

void setup() {

  Serial.begin(9600);

  pinMode(green, OUTPUT);
  pinMode(yellow, OUTPUT);
  pinMode(orange, OUTPUT);
  pinMode(red, OUTPUT);

}

void loop() {

  int value = analogRead(soil);

  Serial.print("Soil Value: ");
  Serial.println(value);

  digitalWrite(green, LOW);
  digitalWrite(yellow, LOW);
  digitalWrite(orange, LOW);
  digitalWrite(red, LOW);

  if (value <= 250) {
    digitalWrite(green, HIGH);
  }
  else if (value <= 500) {
    digitalWrite(yellow, HIGH);
  }
  else if (value <= 750) {
    digitalWrite(orange, HIGH);
  }
  else {
    digitalWrite(red, HIGH);
  }

  delay(300);

}

US1

Ultrasonic Sensor — HC‑SR04

Fires a 10 µs trigger pulse, times the echo return, and converts round‑trip time to distance in centimeters.

TrigD9 → pulse out
EchoD10 ← pulse in
Formuladistance = duration / 58
ultrasonic_hcsr04.ino
int trig = 9;
int echo = 10;

void setup() {

  pinMode(trig, OUTPUT);
  pinMode(echo, INPUT);

  Serial.begin(9600);

}

void loop() {

  digitalWrite(trig, LOW);
  delayMicroseconds(2);

  digitalWrite(trig, HIGH);
  delayMicroseconds(10);

  digitalWrite(trig, LOW);

  long duration = pulseIn(echo, HIGH);

  long distance = duration / 58;

  Serial.print("Distance: ");
  Serial.print(distance);
  Serial.println(" cm");

  delay(500);

}
TEMP1

LM35 Temperature Sensor

Converts the analog voltage to a temperature reading in real time — no external calibration needed.

Sensor checkThe (voltage − 0.5) × 100 formula belongs to a TMP36. A true LM35 outputs 10 mV/°C from 0 V, so use temperature = voltage × 100 instead.
SigA0 ← Vout
Scalevoltage = raw × 5.0 / 1023
As showntemp = (V − 0.5) × 100
lm35_temperature.ino
int tempPin = A0;

void setup() {

  Serial.begin(9600);

}

void loop() {

  int value = analogRead(tempPin);

  float voltage = value * 5.0 / 1023.0;

  float temperature = (voltage - 0.5) * 100;

  Serial.print("Temperature: ");
  Serial.print(temperature);
  Serial.println(" °C");

  delay(1000);

}
IR1

IR Remote Receiver

Decodes incoming infrared frames and prints the received hex code to the serial monitor.

Library versionWritten against the classic IRremote API (decode_results). Library v3.x/v4.x renamed several calls — update accordingly if you're on a recent release.
SigD11 ← receiver OUT
LibIRremote.h
OutputSerial hex on decode
ir_receiver.ino
#include <IRremote.h>

int recv = 11;

IRrecv ir(recv);

decode_results res;

void setup() {

  Serial.begin(9600);

  ir.enableIRIn();

}

void loop() {

  if (ir.decode(&res)) {

    Serial.println(res.value, HEX);

    ir.resume();

  }

}

PIR1

PIR Motion Sensor

Flags motion as a simple digital HIGH, lighting an onboard LED and logging the event to serial.

SigD7 ← PIR OUT
IndicatorD13 → LED
LogicHIGH → motion detected
pir_motion.ino
int pir = 7;
int led = 13;

void setup() {

  pinMode(pir, INPUT);
  pinMode(led, OUTPUT);

  Serial.begin(9600);

}

void loop() {

  int state = digitalRead(pir);

  if (state == HIGH) {

    digitalWrite(led, HIGH);

    Serial.println("Motion Detected!");

  }
  else {

    digitalWrite(led, LOW);

  }

  delay(200);

}
